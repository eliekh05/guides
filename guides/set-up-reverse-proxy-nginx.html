Set Up a Reverse Proxy with Nginx

A reverse proxy sits in front of your app and forwards requests to it. This lets you run apps on internal ports (3000, 8080, etc.) while exposing them on port 80/443 with a domain name, SSL, and extra security.

Install Nginx

# Ubuntu/Debian
sudo apt install nginx -y

# Fedora/RHEL
sudo dnf install nginx -y

# Start and enable
sudo systemctl enable --now nginx

Basic Reverse Proxy Config

Create a new config file for your site:

sudo nano /etc/nginx/sites-available/myapp
server {
    listen 80;
    server_name myapp.example.com;

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }
}
# Enable the site
sudo ln -s /etc/nginx/sites-available/myapp /etc/nginx/sites-enabled/

# Test config syntax
sudo nginx -t

# Reload Nginx
sudo systemctl reload nginx

Add SSL with Let's Encrypt

sudo apt install certbot python3-certbot-nginx -y
sudo certbot --nginx -d myapp.example.com

Certbot automatically updates your Nginx config to add SSL and redirect HTTP to HTTPS.

Proxy Multiple Apps

# /etc/nginx/sites-available/apps

# App 1 — running on port 3000
server {
    listen 443 ssl;
    server_name app1.example.com;
    # ssl_certificate lines added by certbot

    location / {
        proxy_pass http://localhost:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

# App 2 — running on port 8080
server {
    listen 443 ssl;
    server_name app2.example.com;

    location / {
        proxy_pass http://localhost:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

Proxy to a Docker Container

location / {
    proxy_pass http://172.17.0.2:3000;    # Docker container IP
    # OR use container name if on custom network:
    proxy_pass http://mycontainer:3000;
}

WebSocket Support

Apps using WebSockets (Next.js dev, Socket.io, etc.) need these headers:

location / {
    proxy_pass http://localhost:3000;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    proxy_set_header Host $host;
}

Useful Settings

server {
    # Increase timeout for slow apps
    proxy_connect_timeout 60s;
    proxy_send_timeout 60s;
    proxy_read_timeout 60s;

    # Increase upload size limit (default is 1MB)
    client_max_body_size 50M;

    # Gzip compression
    gzip on;
    gzip_types text/plain application/json application/javascript text/css;
}

Check Logs

# Access log
sudo tail -f /var/log/nginx/access.log

# Error log
sudo tail -f /var/log/nginx/error.log

# Logs for a specific site
sudo tail -f /var/log/nginx/myapp.access.log
