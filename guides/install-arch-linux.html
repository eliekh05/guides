Install Arch Linux

Arch Linux has no graphical installer. Installation is done entirely from the command line, which means you understand exactly what is on your system. This guide covers a standard UEFI installation.

Download and Boot

Download the ISO from https://archlinux.org/download/. Flash with Ventoy or dd:

sudo dd if=archlinux-x86_64.iso of=/dev/sdX bs=4M status=progress && sync

Boot from USB. You land at a root shell.

Connect to the Internet

# Ethernet — usually auto-connects
ping archlinux.org

# Wi-Fi
iwctl
[iwd] device list
[iwd] station wlan0 scan
[iwd] station wlan0 get-networks
[iwd] station wlan0 connect "NetworkName"
[iwd] exit

Update System Clock

timedatectl set-ntp true

Partition the Disk

# List disks
fdisk -l
lsblk

# Partition with gdisk (UEFI/GPT)
gdisk /dev/sda

Create these partitions:

PartitionSizeTypeMount
/dev/sda1512 MBEFI System (ef00)/boot
/dev/sda2RemainingLinux filesystem (8300)/
 
# Format
mkfs.fat -F32 /dev/sda1
mkfs.ext4 /dev/sda2     # or mkfs.btrfs /dev/sda2 for Btrfs

# Mount
mount /dev/sda2 /mnt
mount --mkdir /dev/sda1 /mnt/boot

Install Base System

pacstrap -K /mnt base linux linux-firmware base-devel vim networkmanager

Configure the System

# Generate fstab
genfstab -U /mnt >> /mnt/etc/fstab

# Chroot
arch-chroot /mnt

# Set timezone
ln -sf /usr/share/zoneinfo/Region/City /etc/localtime
hwclock --systohc

# Locale
echo "en_US.UTF-8 UTF-8" >> /etc/locale.gen
locale-gen
echo "LANG=en_US.UTF-8" > /etc/locale.conf

# Hostname
echo "myhostname" > /etc/hostname

# Root password
passwd

Install Bootloader

# Install GRUB
pacman -S grub efibootmgr
grub-install --target=x86_64-efi --efi-directory=/boot --bootloader-id=ARCH
grub-mkconfig -o /boot/grub/grub.cfg

Create a User

useradd -m -G wheel -s /bin/bash youruser
passwd youruser

# Enable sudo for wheel group
EDITOR=vim visudo
# Uncomment: %wheel ALL=(ALL:ALL) ALL

Enable NetworkManager

systemctl enable NetworkManager

Reboot

exit          # exit chroot
umount -R /mnt
reboot

Remove the USB drive when the screen goes blank.

Post-Install

# Connect to Wi-Fi
nmcli device wifi connect "NetworkName" password "password"

# Install yay (AUR helper)
sudo pacman -S git
git clone https://aur.archlinux.org/yay-bin.git
cd yay-bin && makepkg -si

# Install a desktop environment (optional)
sudo pacman -S gnome gnome-extra
sudo systemctl enable gdm

# Or KDE
sudo pacman -S plasma kde-applications
sudo systemctl enable sddm
